cmake_minimum_required(VERSION 3.20)
project(CrashNativeInspector VERSION 0.1 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "CrashNativeInspector currently targets Windows")
endif()
add_executable(CrashNativeInspector WIN32 src/inspector.cpp)
target_compile_features(CrashNativeInspector PRIVATE cxx_std_17)
target_compile_definitions(CrashNativeInspector PRIVATE UNICODE _UNICODE)
target_link_libraries(CrashNativeInspector PRIVATE comdlg32 gdi32 user32)
if(MSVC)
  target_compile_options(CrashNativeInspector PRIVATE /W4 /EHsc)
endif()
